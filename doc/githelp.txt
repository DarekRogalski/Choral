# dodanie plików do projektu, dodanie zmian w projekcie (przenieś plik do przechowalni)
git add .
git add plik.tex

# commit czyli zapisanie zmian w git (stworzenie migawki zmian w projekcie) LOKALNIE
git commit -m "informacja o zmianach"

# opcja -am tworzy commit wszystkich zmian w przechowalni:
git commit -am "informacja o zmianach"

# sprawdzenie efektu dodawania plików/zmian
git status

# wykaz różnic w zestawach danych wejściowych (wykazuje wszystkie niezatwierdzone zmiany od ostatniego commitu)
git diff

# odłożenie tymczasowo zmian w pliku na półkę, gdy nie jest gotowy, aby rozpocząć pracę nad innym (zmian kontekstu) - dot. plików które zostały dodane git add
git stash

# ponowne zastosowanie zmian, odłożonych na półkę
git stash pop

# ignorowanie wcześniej zatwierdzonego pliku:
echo debug.log >> .gitignore
git rm --cached debug.log
git commit -m "zacznij ignorować debug.log"

# dodanie pliku ignorowanego
git add -f debug.log
git commit -m "wymuś dodanie debug.log"

# sprawdzenie pliku ignorowanego, która reguła go ignoruje
git check-ignore -v debug.log

# stan katalogu roboczego i przechowalni (nie zawiera historii commitów)
git status

# stan kat. Roboczego i historii commitów, 
git log

# opcja -n <limit> wyświetli ostatnie n commitów
git log -n 3

# jaki kod został dodany do repozytorium, dlaczego i w jaki sposób
git blame
